% --- Section II.D [69]-[74], Table 3: direct and inverse form -------------
% Hand check: Q = P - 50 <=> P = 50 + Q; Q = 85 - 0.5P <=> 0.5P = 85 - Q
% <=> P = 170 - 2Q. Slopes: demand dQ/dP = -0.5, dP/dQ = -2 = 1/(-0.5);
% supply 1 and 1.
\begin{frame}[plain,label=w2_01_forms]
  \Large
  \[
    \tag{1}\;\text{Direct form: } Q^S(P), \quad Q^D(P)
  \]
  \[
    \tag{2}\;\text{Inverse form: } P^S(Q), \quad P^D(Q)
  \]
  \[
    \tag{3}\;Q^S = P - 50 \;\Leftrightarrow\; P = 50 + Q
  \]
  \[
    \tag{4}\;Q^D = 85 - 0.5P \;\Leftrightarrow\; P = 170 - 2Q
  \]

  \vspace{0.5em}
  \begin{itemize}\Large
    \item Valid for $P \ge 50$ (supply), $P \le 170$ (demand)
    \item Inverse slope $= 1 /$ direct slope
    \item Demand: $-0.5$ and $-2$; supply: $1$ and $1$
  \end{itemize}
\end{frame}

% --- Section II.E [76]: derivative of a quadratic ---------------------------
\begin{frame}[plain,label=w2_02_deriv]
  \Large
  \[
    \tag{5}\;TC(q) = a + bq + cq^2
  \]
  \[
    \tag{6}\;MC(q) = \frac{dTC}{dq} = b + 2cq
  \]

  \vspace{0.75em}
  \begin{itemize}\Large
    \item $a$: fixed cost, disappears in the derivative
    \item $bq$ contributes $b$; $cq^2$ contributes $2cq$
    \item $MC$: the cost of one more unit
  \end{itemize}
\end{frame}

% --- Section II.E [77]-[78]: from marginal to total, producer surplus -------
\begin{frame}[plain,label=w2_03_integral]
  \Large
  \[
    \tag{7}\;TVC(q) = \int_0^{q} MC(x)\,dx = bq + cq^2
  \]
  \begin{align*}
    \tag{8}\;PS &= TR - TVC = P \times q - TVC(q)\\
                &= \int_0^{q} \bigl[P - MC(x)\bigr]\,dx
  \end{align*}

  \vspace{0.5em}
  \begin{itemize}\Large
    \item Integral $=$ area under $MC$, $0$ to $q$
    \item $PS$: between price line and $MC$
  \end{itemize}
\end{frame}

% --- Section II.E [79]: triangles, CS and PS -------------------------------
\begin{frame}[plain,label=w2_04_triangle]
  \Large
  \begin{align*}
    \tag{9}\;\text{Area of a triangle} &= \tfrac{1}{2} \times \text{base}\\
                                       &\phantom{=}\; \times \text{height}
  \end{align*}
  \[
    \tag{10}\;CS = \tfrac{1}{2}\, Q^* \,(\text{demand intercept} - P^*)
  \]
  \[
    \tag{11}\;PS = \tfrac{1}{2}\, Q^* \,(P^* - \text{supply intercept})
  \]

  \vspace{0.75em}
  \begin{itemize}\Large
    \item Base $= Q^*$ on the quantity axis
    \item Height $=$ vertical distance on the price axis
  \end{itemize}
\end{frame}

% --- Section II.E [80]: area under a straight line (Question 3.4d) ---------
% Hand check: 220*40 - 0.5*40^2 = 8800 - 800 = 8000;
% 50*40 + 0.5*40^2 = 2000 + 800 = 2800
% A, b and u, s are the intercepts and slopes of the two lines (rectangle
% minus or plus a triangle), as in Worked example 6.
\begin{frame}[plain,label=w2_05_arealine]
  \Large
  \[
    \tag{12}\;\int_0^{Q} (A - bx)\,dx = AQ - \tfrac{1}{2} b Q^2
  \]
  \[
    \tag{13}\;\int_0^{Q} (u + sx)\,dx = uQ + \tfrac{1}{2} s Q^2
  \]
  \[
    \tag{14}\;220 \times 40 - \tfrac{1}{2} \times 40^2 = 8{,}800 - 800 = 8{,}000
  \]
  \[
    \tag{15}\;50 \times 40 + \tfrac{1}{2} \times 40^2 = 2{,}800
  \]

  \vspace{0.3em}
  \begin{itemize}\Large
    \item Under $P = 220 - Q$ and $MC = 50 + Q$
    \item Needed in Question 3.4d
  \end{itemize}
\end{frame}

% --- Worked example 1, step 1 [96]-[98]: the firm's supply function ---------
% Hand check: P = 50 + 10q => 10q = P - 50 => q = (P - 50)/10; q(50) = 0
\begin{frame}[plain,label=w2_07_we1a]
  \Large
  \begin{align*}
    \tag{19}\;P &= MC(q) = 50 + 10q\\
    \Rightarrow 10q &= P - 50\\
    \Rightarrow q(P) &= \frac{P - 50}{10}, \quad P \ge 50
  \end{align*}

  \vspace{0.5em}
  \begin{itemize}\Large
    \item Firm's supply curve $=$ its $MC$ curve
    \item Direct form: $q$ as a function of $P$
    \item $P < 50$: no supply (first unit costs $50$)
  \end{itemize}
\end{frame}

% --- Worked example 1, steps 2-3 [99]-[102]: the industry supply curve ------
% Hand check: Q = 10*(P - 50)/10 = P - 50 => P = 50 + Q; firm slope 10,
% industry slope 1 (+1 pound: +0.1 per firm, +1 for 10 firms)
\begin{frame}[plain,label=w2_08_we1b]
  \Large
  \begin{align*}
    \tag{20}\;Q_N(P) &= 10 \times q(P) = 10 \times \frac{P - 50}{10}\\
                    &= P - 50
  \end{align*}
  \[
    \tag{21}\;P_N = 50 + Q_N
  \]

  \vspace{0.5em}
  \begin{itemize}\Large
    \item Horizontal sum: add quantities, not prices
    \item Slope: firm $10$, industry $1$
    \item A \pounds 1 rise: $+0.1$ per firm, $+1$ in total
  \end{itemize}
\end{frame}

% --- Worked example 2 [107]-[110]: the innovation shifts supply -------------
% Hand check: MC = 20 + 10q; q = (P - 20)/10 (P >= 20); Q = P - 20,
% P = 20 + Q. Vertical shift 50 - 20 = 30 at every Q; horizontal:
% industry (P - 20) - (P - 50) = 30, firm 30/10 = 3
\begin{frame}[plain,label=w2_09_we2]
  \Large
  \[
    \tag{22}\;MC(q) = \frac{dTC_N}{dq} = 20 + 10q
  \]
  \[
    \tag{23}\;q(P) = \frac{P - 20}{10}, \quad P \ge 20
  \]
  \[
    \tag{24}\;Q_N = P - 20 \;\Leftrightarrow\; P_N = 20 + Q_N
  \]

  \vspace{0.5em}
  \begin{itemize}\Large
    \item Down by $30$ at every $Q$ (the cost change)
    \item Right by $30$ units (industry, slope $1$)
    \item Right by $3$ units (firm, slope $10$)
  \end{itemize}
\end{frame}

% --- Technical box [137]-[141]: definition of the price elasticity ---------
\begin{frame}[plain,label=w2_13_elast_def]
  \Large
  \begin{align*}
    \tag{40}\;\varepsilon_D &= \frac{\%\Delta Q}{\%\Delta P}
      = \frac{\Delta Q / Q}{\Delta P / P}\\
    &= \frac{\Delta Q}{\Delta P} \times \frac{P}{Q}
  \end{align*}
  \[
    \tag{41}\;\varepsilon_D = \frac{dQ}{dP} \times \frac{P}{Q}
    \qquad \text{(point elasticity)}
  \]

  \vspace{0.5em}
  \begin{itemize}\Large
    \item $dQ/dP$: slope of the direct form $Q(P)$
    \item $P/Q$ cancels the units: a pure number
  \end{itemize}
\end{frame}

% --- Technical box [142], [144]: reading the number, slope vs elasticity ---
% Check of the slope link at E0 = (40, 90): P/Q = 2.25; b = 2.25/1.125 = 2;
% s = 2.25/2.25 = 1 (the nickel slopes)
\begin{frame}[plain,label=w2_14_elast_read]
  \Large
  \[
    \tag{42}\;\varepsilon_S = \frac{dQ^S}{dP} \times \frac{P}{Q} > 0
  \]
  \[
    \tag{43}\;b = \frac{P/Q}{|\varepsilon_D|}, \qquad s = \frac{P/Q}{\varepsilon_S}
  \]

  \vspace{0.3em}
  \begin{itemize}\Large
    \item $b, s$: slopes of the inverse curves
    \item $|\varepsilon_D|$: elastic $> 1$, unit $= 1$, inelastic $< 1$
    \item Vertical demand: $\varepsilon_D = 0$ ($b$ infinite)
    \item Horizontal demand: $\varepsilon_D \to -\infty$ ($b = 0$)
  \end{itemize}
\end{frame}

% --- Technical box [143]: constant slope, changing elasticity --------------
% Hand check: P = 20: Q = 100 - 40 = 60, eps = -2*20/60 = -0.667;
% P = 40: Q = 100 - 80 = 20, eps = -2*40/20 = -4
\begin{frame}[plain,label=w2_15_elast_example]
  \Large
  \[
    \tag{44}\;Q = 100 - 2P \;\Rightarrow\; \frac{dQ}{dP} = -2
  \]
  \[
    \tag{45}\;\varepsilon_D\big|_{P = 20} = -2 \times \frac{20}{60} = -0.67
  \]
  \[
    \tag{46}\;\varepsilon_D\big|_{P = 40} = -2 \times \frac{40}{20} = -4
  \]

  \vspace{0.3em}
  \begin{itemize}\Large
    \item $P = 20$: $Q = 60$ (inelastic)
    \item $P = 40$: $Q = 20$ (elastic)
    \item One slope, a new elasticity at every point
  \end{itemize}
\end{frame}
